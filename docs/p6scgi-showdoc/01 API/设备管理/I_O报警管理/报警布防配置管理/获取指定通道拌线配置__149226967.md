# 获取指定通道拌线配置
##### 简要描述

- ##### 说明

URI中的<ChannelID>显示通道编号。 <ChannelID>从1开始。

##### 请求

**xml**
无
**字段说明**
无

##### 响应

**xml**

```
<AssisstantLineConfig>
    <ChannelID><!--ro,req,xs:integer--></ChannelID>
    <Enable><!--rw,req,xs:string,”false”,”true”--></Enable>
    <Line>
        <StartPosX><!--rw,req,xs:interger;(0-703)--></StartPosX>
        <StartPosY><!--rw,req,xs:interger;(0-575)--></StartPosY>
        <EndPosX><!--rw,req,xs:interger;(0-703)--></EndPosX>
        <EndPosY><!--rw,req,xs:interger;(0-575)--></EndPosY>
        <Direction><!--rw,req,xs:string,”forward_direction”,”opposite_direction”,”two_way”--></Direction>
    </Line>
    <Sensitivity><!--rw,req,xs:string,”highest”,”higher”,”high”,”middle”,”low”,”lower”,”lowest”--></Sensitivity>
    <AssisstantLineUseful>
        <UseForPeopleDetect><!--rw,req,xs:string,”false”,”true”--></UseForPeopleDetect>
    </AssisstantLineUseful>
    <Trigger>
            <NotifyAMS>
                <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
            </NotifyAMS>
            <BeepAlert>
                <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
            </BeepAlert>
            <FullScreen>
                <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
            </FullScreen>
            <Mail>
                    <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
            </Mail>
            <Ftp>
                    <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
            </Ftp>
            <Push>
                    <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
            </Push>
            <AlarmOut>
                <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
                <AlarmOutMask><!--rw,req,xs:string,” 111111111111111”--></AlarmOutMask>
            </AlarmOut>
            <Snapshot>
                <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
                <SnapshotMask><!--rw,req,xs:string,” 111111111111111”--></SnapshotMask>
            </Snapshot>
            <Record>
                <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
                <RecordMask><!--rw,req,xs:string,” 111111111111111”--></RecordMask>
            </Record>
            <PTZ>
                <Enable><!--rw,req,xs:boolean,”false”,”true”--></Enable>
                <PTZActionList>
                    <ptzChannelID> <!--rw,req, xs:string --></ptzChannelID>
                    <ActionName><!-- rw,req, xs:string,”none,preset, pattern, patrol” --></ActionName>
                    <ActionNum><!--rw,req,xs:integer,depend on <ActionName>--></ActionNum>
                </PTZActionList>
            </PTZ>
    </Trigger>
</AssisstantLine>
```
