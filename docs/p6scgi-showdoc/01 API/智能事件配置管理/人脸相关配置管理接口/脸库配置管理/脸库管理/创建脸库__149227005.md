# 创建脸库
##### 简要描述

- #### 以T5AI设备为例说明

##### **请求**

**_xml_**

```
<?xml version="1.0" encoding="utf-8"?>
<FaceGroup>
    <GroupName><!-- wo,req:string, the max length 32 --></GroupName>
    <GroupID2><!-- rw,req:string, the max length 64 --></GroupID2>
    <Ownner><!-- wo,req:string, the max length 64 --></Ownner>
    <SystemTime><!-- wo,req:string, the max length 32 --></SystemTime>
    <GroupThresholdValue><!-- wo,opt:float, range [0,100] --></GroupThresholdValue>
</FaceGroup>
```

**_字段说明_**

| 字段 | 类型 | 功能描述 | 是否必须 | 字段是否有效 | 是否可为空 | 可读可写(rw)/只读(ro)/只写(wo) | 取值范围/长度 | 备注 | 在APP的哪一个页面的哪一个字段 | 在web的哪一个页面的哪一个字段 | 字段受哪一个能力限制 | 字段受哪个字段限制 |
| --- | --- | ---- | ---- | ------ | ----- | ---------------------- | ------- | --- | ---------------- | ---------------- | ---------- | --------- |
| FaceGroup |  |  |  |  |  |  |  |  |  | web没有调用接口 |  |  |
| GroupName | string | <span class="colour" style="color:rgb(106, 153, 85)">脸库名字</span> | 是 | 是 | 不可为空 | 只写 | 最大长度32 | 注意避免脸库名称重复 |  |  |  |  |
| GroupID2 | string | <span class="colour" style="color:rgb(106, 153, 85)">脸库ID，可使用32位的uuid，保证唯一性即可</span> | 是 | 是 | 不可为空 | 可读可写 | 最大长度64 |  |  |  |  |  |
| Ownner | string | <span class="colour" style="color:rgb(106, 153, 85)">安全码， 通常和SystemTime一起传输给设备，由拥有者(设备脸库的拥有者)和系统时间进行相关计算</span> | 是 | 是 | 不可为空 | 只写 | 最大长度64 | <span class="colour" style="color:rgb(106, 153, 85)"><span class="colour" style="color:rgb(106, 153, 85)">参考“安全码生成源码”</span></span> |  |  |  |  |
| SystemTime | string | <span class="colour" style="color:rgb(106, 153, 85)">系统时间，脸库拥有者使用SystemTime生成Ownner, 格式：YYYY-MM-DDThh:mm:ss</span> | 是 | 是 | 不可为空 | 只写 | 最大长度32 |  |  |  |  |  |
| GroupThresholdValue | float | 脸库阈值，只有抓拍的人脸阈值大于脸库阈值的人脸会认为对比通过 | 否 | 是 | 不可为空 | 只写 | [0,100] |  |  |  |  |  ||

**安全码生成源码**

``` java
//TIP 要<b>运行</b>代码，请按 <shortcut actionId="Run"/> 或
// 点击装订区域中的 <icon src="AllIcons.Actions.Execute"/> 图标。
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
 
public class Main {
 
/**
* 获取当前系统时间并返回格式化后的字符串
*
* @param format 时间格式，例如 "yyyy-MM-dd'T'HH:mm:ss"
* @return 格式化后的系统时间字符串
*/
public static String getFormattedSystemTime(String format) {
// 默认格式
String dateTimeFormat = (format == null || format.isEmpty()) ? "yyyy-MM-dd'T'HH:mm:ss" : format;
 
LocalDateTime now = LocalDateTime.now();
DateTimeFormatter formatter = DateTimeFormatter.ofPattern(dateTimeFormat);
return now.format(formatter);
}
 
private static String getUniqueCodeString() {
// 模拟获取序列号
return "1234567890ABCDEF";
}
 
 
public static int GenerateSafetyCode(String uniqueCode,String systemTime, StringBuilder pSafetyCode) {
int IDVR_OK = 0;
 
if (pSafetyCode == null || pSafetyCode.capacity() <= 0) {
return -1;
}
 
try {
// 计算 MD5
String srcBuf = uniqueCode + systemTime;
MessageDigest md = MessageDigest.getInstance("MD5");
byte[] md5Bytes = md.digest(srcBuf.getBytes());
 
// 转换 MD5 为字符串
StringBuilder md5Str = new StringBuilder();
for (byte b : md5Bytes) {
md5Str.append(String.format("%02x", b));
}
 
// 截取前8位并反转前4位和后4位
String partialMd5 = md5Str.substring(0, 8);
char[] chars = partialMd5.toCharArray();
for (int i = 0; i < 4; i++) {
char temp = chars[i];
chars[i] = chars[7 - i];
chars[7 - i] = temp;
}
 
// 写入安全码
String finalCode = new String(chars);
pSafetyCode.replace(0, pSafetyCode.capacity(), finalCode);
 
return IDVR_OK;
 
} catch (NoSuchAlgorithmException e) {
e.printStackTrace();
return -1;
}
}
 
 
public static void main(String[] args) {
 
String systemTime = getFormattedSystemTime("yyyy-MM-dd'T'HH:mm:ss");
System.out.println("System Time: " + systemTime);
 
//获取唯一码字符串
// 模拟获取设备序列号
String uniqueCode = getUniqueCodeString();
System.out.println("Unique Code: " + uniqueCode);
 
 
//生成安全码
StringBuilder pSafetyCode = new StringBuilder(64);
GenerateSafetyCode(uniqueCode,systemTime, pSafetyCode);
System.out.println("Safety Code: " + pSafetyCode);
}
}
```
