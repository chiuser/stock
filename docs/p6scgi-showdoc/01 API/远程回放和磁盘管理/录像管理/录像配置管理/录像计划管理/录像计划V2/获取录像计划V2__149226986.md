            <ID><!--rw,req,xs:integer; "0,1,2...",max value 15--></ID>
            <Enable><!--rw,req,xs:boolean”true”,”false”--></Enable>
            <StartTime><!--rw,req,xs:string; max size 32,格式hh:mm,24小时制--></StartTime>
