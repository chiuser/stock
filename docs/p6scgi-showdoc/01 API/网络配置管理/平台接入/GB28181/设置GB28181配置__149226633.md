# 设置GB28181配置
##### 简要描述

- ##### 请求

**xml**

```

<GB28181Cfg>
    <Enable><!--rw,opt,xs:boolean--></Enable>
    <ProtocolVersion><!--rw,opt,xs:string,"2011","2016","2022"--></ProtocolVersion>
    <HeartbeatInterval><!--rw,opt,xs:integer,the unit is second,5-255--></HeartbeatInterval>
    <MaxHeartbeatTimeoutCount><!--rw,opt,xs:integer,the unit is time,3-255--></MaxHeartbeatTimeoutCount>
    <RegisterExpire><!--rw,opt,xs:integer,the unit is second,3600-100000--></RegisterExpire>
    <RegisterInterval><!--rw,opt,xs:integer,the unit is second,60-3600--></RegisterInterval>
    <RegisterState><!--rw,opt,xs:string,"offline","online"--></RegisterState>
    <StreamType><!--rw,opt,xs:string,"main","sub","second"--></StreamType>
    <VoiceTalkTransportType><!--rw,opt,xs:string,"udp","tcpActive","tcpPassive"--></VoiceTalkTransportType>
	<DevieceSIPPort><!--rw,req,xs:integer--></DevieceSIPPort>
    <ServerCfg>
        <ServerID><!--rw,req,xs:string,the length must be 20--></ServerID>
        <ServerRegion><!--rw,req,xs:string,the length must be 10--></ServerDomain>
        <ServerIpAddress><!--rw,req,xs:string,ip or domain--></ServerIpAddress>
        <ServerPort><!--rw,req,xs:integer--></ServerPort>
        <UserName><!--rw,req,xs:string, length : 32--></UserName>
        <UserAuthID><!--rw,req,xs:string, length : 32--></UserAuthID>
        <Password><!--rw,req,xs:string, length : 32--></Password>
        <TransportLayType><!--rw,req,xs:string,"tcp","udp"--></TransportLayType>
    </ServerCfg>
    <AlarmInInfoList>
        <AlarmInInfo>
            <ID><!--rw,req,xs:integer--></ID>
            <AlarmInID><!--rw,req,xs:string,the length must be 20--></AlarmInID>
        </AlarmInInfo>
        …
    </AlarmInInfoList>
    <VideoOutInfoList>
        <VideoOutInfo>
            <ID><!--rw,req,xs:integer--></ID>
            <VideoOutID><!--rw,req,xs:string,the length must be 20--></VideoOutID>
        </VideoOutInfo>
        …
    </VideoOutInfoList>
    <AudioOutInfoList>
        <AudioOutInfo>
            <ID><!--rw,req,xs:integer--></ID>
            <AudioOutID><!--rw,req,xs:string,the length must be 20--></AudioOutID>
        </AudioOutInfo>
        …
    </AudioOutInfoList>
</GB28181Cfg>
```
