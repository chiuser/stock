```
<AudioStream>

    <AudioInCodec><!—-rw,req,xs:string,”G.711U”--></AudioInCodec>
    <AudioInVolume><!—-rw,req,xs:integer,”0--100”--></AudioInVolume>
    <AudioOutVolume><!—-rw,opt,xs:integer,”0--100”--></AudioOutVolume>
    <EnableAlarmAudio><!—-rw,opt,xs:boolean--></EnableAlarmAudio>
    <EnableAlarmFlashLight><!—-rw,opt,xs:boolean--></EnableAlarmFlashLight>
    <AlarmVolume><!—-rw,opt,xs:integer--></AlarmVolume>
    <VoiceType><!—-rw,opt,xs:string--></VoiceType>
    <AlarmDuration><!—-rw,opt,xs:integer--></AlarmDuration>
    <AlarmPlayTimes><!—-rw,opt,xs:integer--></AlarmPlayTimes>
    <AlarmSchedule Version="1.0">
        <AllDay><!--rw,req,xs:boolean--></AllDay>
        <TimeBlockList Version="1.0">
            <TimeBlock_0>111111111111111111111111111111111111111111111111</TimeBlock_0>
            <TimeBlock_1>111111111111111111111111111111111111111111111111</TimeBlock_1>
            <TimeBlock_2>111111111111111111111111111111111111111111111111</TimeBlock_2>
            <TimeBlock_3>111111111111111111111111111111111111111111111111</TimeBlock_3>
            <TimeBlock_4>111111111111111111111111111111111111111111111111</TimeBlock_4>
            <TimeBlock_5>111111111111111111111111111111111111111111111111</TimeBlock_5>
            <TimeBlock_6>111111111111111111111111111111111111111111111111</TimeBlock_6>
        </TimeBlockList>
    </AlarmSchedule>
    <AlarmTimeSlotList Version="1.0">
        <SchedTimeSlot>
            <ID><!--rw,req,xs:interger, 1,2,3...--></ID>
            <Enable><!--rw,req,xs:boolean,"true","false"--></Enable>
            <StartTime><!--rw,req,xs:string,"hh:mm",max length 32--></StartTime>
            <EndTime><!--rw,req,xs:string,"hh:mm",max length 32--></EndTime>
            <Weekday><!--rw,req,xs:string,"11111111",max length 8; 每周7天, 字符串内容从左至右分别表示周日,周一...,以此类推;取值1表示选定对应星期,取值0表示没有选择对应星期,第8位保留--></Weekday>
        </SchedTimeSlot>
        ...
    </AlarmTimeSlotList>
</AudioStream>
```
