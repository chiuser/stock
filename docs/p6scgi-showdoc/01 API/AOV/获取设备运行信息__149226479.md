无

            <ID><!--ro,req,xs:interger--></ID>

            <PeopleDetectCount><!--ro,req,xs:interger--></PeopleDetectCount>
        </PeopleDetectCountDay>
        ...
    </IntradayWeekPeopleDetectCount>
</DeviceRunningInfo>
