	int iRegisterInterval;      	// 注册间隔
	int iKeepAliveInterval;       	// 保活间隔
	int iProtocolType;            	// 协议类型；0.http，1.https
    
    char cReserve[256];        // 保留字段
}HY_DVR_P6SGA1400_CONFIG, *LPHY_DVR_P6SGA1400_CONFIG;

typedef struct tagHY_DVR_P6SGA1400_SERVER_CONNECT_STATUS
{
    int bValid;

    char cReserve[256];        // 保留字段
}HY_DVR_P6SGA1400_SERVER_CONNECT_STATUS, *LPHY_DVR_P6SGA1400_SERVER_CONNECT_STATUS;

typedef struct tagHY_DVR_P6SGA1400_SERVER_CONNECT_TEST

    char acTime[32];                            /*!< 输出参数,服务器响应时间 */
    char acInfo[1024];                          /*!< 输出参数,服务器返回的信息(使用base64编码) */

    char acOutputReserved[256];        /*!< 输出参数保留字段 */
}HY_DVR_P6SGA1400_SERVER_CONNECT_TEST, *LPHY_DVR_P6SGA1400_SERVER_CONNECT_TEST;
