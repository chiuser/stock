# 中间层主次命令及结构体
##### 简要描述


**xml**

```

//主命令
//获取云消息
#define HY_DVR_GET_CLOUD_MESSAGE        1165
//设置云消息
#define HY_DVR_SET_CLOUD_MESSAGE        1166

//次命令
//获取消息错误码列表(HY_DVR_CLOUD_MESSAGE_ERROR_CODE_LIST)
#define CLOUD_MESSAGE_ERROR_CODE_LIST   1


//云消息错误码
typedef struct tagHY_DVR_CLOUD_MESSAGE_ERROR_CODE_INFO
{
      int nValid;    //本结构体是否有效

      int iID;                   /* 错误码ID,从1开始递增,最大值256 */
      int iThreadID;             /* 线程ID */
      char acThreadName[16];     /* 线程名,最大长度16 */
      int iErrorCode;            /* 云消息错误码 */
}HY_DVR_CLOUD_MESSAGE_ERROR_CODE_INFO,*LPHY_DVR_CLOUD_MESSAGE_ERROR_CODE_INFO;

#define MAX_ERROR_CODE_LIST 256 //错误码列表最大个数
//云消息错误码列表
typedef struct tagHY_DVR_CLOUD_MESSAGE_ERROR_CODE_LIST
{
    int nValid;

    int listCount; //当前list个数,最大值256
    HY_DVR_CLOUD_MESSAGE_ERROR_CODE_INFO stErrorCodeList[MAX_ERROR_CODE_LIST]; //错误码列表
}HY_DVR_CLOUD_MESSAGE_ERROR_CODE_LIST,*LPHY_DVR_CLOUD_MESSAGE_ERROR_CODE_LIST;
```
