### 错误码统一定义规则

    /******************************************************************************
     * 错误代码定义: 类型为SINT32,错误代码必须是负数,所以最高位置1,32位定义如下:
     *  +-+-----+-------+---------+-----+---------------+---------------+
     *  |1|0 9 8|7 6 5 4|3 2 1 0 9|8 7 6|5 4 3 2 1 0 9 8|7 6 5 4 3 2 1 0|
     *  +-+-----+-------+---------+-----+---------------+---------------+
     *  |1|LEVEL|   R   | PROTOID |WHERE|            ERRORNO            |
     *  +-+-----+-------+---------------+---------------+---------------+
     * 这里: LEVEL:错误级别,R:保留位,PROTOID:协议ID,WHERE:错误地方,ERRORNO:错误号
     * 当返回错误时,采用ERRNO_R(Level,Id,Where,Error)形式
     *****************************************************************************/
    #define    GET_LEVEL(ErrNo)    (((ErrNo)>>28) & 0x7)
    #define    GET_LIBTYPE(ErrNo)    (((ErrNo)>>27) & 0x1)
    #define    GET_PROTOID(ErrNo)    (((ErrNo)>>19) & 0x1F)
    #define    GET_WHERE(ErrNo)    (((ErrNo)>>16) & 0x7)
    #define    GET_ERRNO(ErrNo)    ((ErrNo) & 0x0000FFFF)
    
    /*
    *错误级别[LEVEL]
    *ERROR_LEVEL_MODULE:单独某一模块重启后就可以解决该问题
    *ERROR_LEVEL_SYSTEM:整个应用软件系统重启才可以解决该问题
    *ERROR_LEVEL_OS:机器重启才可以解决该问题
    */
    #define    ERROR_LEVEL_MODULE    1    /* 模块级别错误 */
    #define    ERROR_LEVEL_SYSTEM    2    /* 系统级别错误 */
    #define    ERROR_LEVEL_OS        3    /* 操作系统级别错误 */

    *库类型[L]

    #define    LIBTYPE_CLIENT    0
